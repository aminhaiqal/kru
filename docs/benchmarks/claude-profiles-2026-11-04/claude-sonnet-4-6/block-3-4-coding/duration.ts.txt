export function parseDuration(text: string): number {
  const MAX = Number.MAX_SAFE_INTEGER; // 9007199254740991

  // Unit name → rank (strictly ascending h < m < s < ms)
  const unitRank: Record<string, number> = { h: 0, m: 1, s: 2, ms: 3 };
  // Unit name → ms multiplier
  const unitMult: Record<string, number> = { h: 3600000, m: 60000, s: 1000, ms: 1 };
  // Try 'ms' before 'm' to avoid 'm' consuming the 'm' in 'ms'
  const tryOrder = ['h', 'ms', 'm', 's'];

  const trimmed = text.trim();
  if (!trimmed) throw new RangeError('Invalid duration');

  let pos = 0;
  let total = 0;
  let lastRank = -1;
  let count = 0;

  while (pos < trimmed.length) {
    // Skip whitespace between components (but not between digits and unit)
    while (pos < trimmed.length && trimmed[pos] === ' ') pos++;
    if (pos >= trimmed.length) break;

    // Parse nonnegative integer digits (no sign, no dot)
    const digitStart = pos;
    while (pos < trimmed.length && trimmed[pos] >= '0' && trimmed[pos] <= '9') pos++;
    if (pos === digitStart) throw new RangeError('Invalid duration');

    const numStr = trimmed.slice(digitStart, pos);

    // Guard against astronomically long digit strings before Number conversion
    if (numStr.length > 16) throw new RangeError('Invalid duration');

    // Match a unit immediately after the digits (no whitespace allowed here)
    let unit: string | null = null;
    for (const u of tryOrder) {
      if (trimmed.startsWith(u, pos)) { unit = u; break; }
    }
    if (!unit) throw new RangeError('Invalid duration');
    pos += unit.length;

    // Enforce strictly descending h > m > s > ms order (no duplicates, no reversal)
    const rank = unitRank[unit];
    if (rank <= lastRank) throw new RangeError('Invalid duration');
    lastRank = rank;

    // Check the numeric value itself
    const val = Number(numStr);
    if (val > MAX) throw new RangeError('Invalid duration');

    // Check that val * multiplier won't exceed MAX (avoids float overflow)
    const mult = unitMult[unit];
    if (val > Math.floor(MAX / mult)) throw new RangeError('Invalid duration');
    const contribution = val * mult;

    // Check running total
    if (total > MAX - contribution) throw new RangeError('Invalid duration');
    total += contribution;

    count++;
  }

  if (count === 0) throw new RangeError('Invalid duration');
  return total;
}
