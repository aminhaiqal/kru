export function parseDuration(text: string): number {
  const MAX = Number.MAX_SAFE_INTEGER;
  const MULT: Record<string, number> = { h: 3600000, m: 60000, s: 1000, ms: 1 };
  const ORDER: Record<string, number> = { h: 3, m: 2, s: 1, ms: 0 };

  // Validate overall structure: optional whitespace, then one or more <digits><unit>
  // separated by whitespace, then optional whitespace. Units: h, ms, m, s (ms before m).
  if (!/^\s*\d+(h|ms|m|s)(\s*\d+(h|ms|m|s))*\s*$/.test(text)) {
    throw new RangeError('Invalid duration');
  }

  let total = 0;
  let lastOrder = Infinity;
  const componentRe = /(\d+)(h|ms|m|s)/g;
  let match: RegExpExecArray | null;

  while ((match = componentRe.exec(text)) !== null) {
    const unit = match[2];
    const order = ORDER[unit];

    // Units must appear in strictly descending order (no duplicates).
    if (order >= lastOrder) throw new RangeError('Invalid duration');
    lastOrder = order;

    const value = Number(match[1]);
    if (value > MAX) throw new RangeError('Invalid duration');

    const contribution = value * MULT[unit];
    if (contribution > MAX) throw new RangeError('Invalid duration');

    total += contribution;
    if (total > MAX) throw new RangeError('Invalid duration');
  }

  return total;
}
