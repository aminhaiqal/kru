export function parseDuration(text: string): number {
  const MAX = Number.MAX_SAFE_INTEGER;
  const unitOrder = ['h', 'm', 's', 'ms'] as const;
  const multiplier: Record<string, number> = { h: 3600000, m: 60000, s: 1000, ms: 1 };
  // Max component value such that value * multiplier <= MAX
  const maxComp: Record<string, number> = {
    h: Math.floor(MAX / 3600000),
    m: Math.floor(MAX / 60000),
    s: Math.floor(MAX / 1000),
    ms: MAX,
  };

  // Match digits immediately followed by a unit; ms before m to avoid partial match
  const componentRe = /^(\d+)(h|ms|s|m)/;

  let s = text.replace(/^\s+/, '');
  if (!s) throw new RangeError('Invalid duration');

  let total = 0;
  let lastUnitIdx = -1;
  let count = 0;

  while (s.length > 0) {
    const m = componentRe.exec(s);
    if (!m) throw new RangeError('Invalid duration');

    const unit = m[2];
    const uIdx = unitOrder.indexOf(unit as typeof unitOrder[number]);

    // Units must appear in strictly descending order (no duplicates, no out-of-order)
    if (uIdx <= lastUnitIdx) throw new RangeError('Invalid duration');
    lastUnitIdx = uIdx;

    const value = Number(m[1]);

    // Reject component values that would overflow when multiplied
    if (value > maxComp[unit]) throw new RangeError('Invalid duration');

    const contrib = value * multiplier[unit];

    // Reject if running total would exceed MAX_SAFE_INTEGER
    if (total > MAX - contrib) throw new RangeError('Invalid duration');
    total += contrib;
    count++;

    // Advance past matched component, strip optional whitespace
    s = s.slice(m[0].length).replace(/^\s+/, '');
  }

  if (count === 0) throw new RangeError('Invalid duration');
  return total;
}
