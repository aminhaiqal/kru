const PATTERN =
  /^\s*(?:(\d+)h)?\s*(?:(\d+)m(?!s))?\s*(?:(\d+)s)?\s*(?:(\d+)ms)?\s*$/;

const MULTIPLIERS = [3_600_000n, 60_000n, 1_000n, 1n];

export function parseDuration(text: string): number {
  const match = typeof text === 'string' ? PATTERN.exec(text) : null;
  if (!match || match.slice(1).every((part) => part === undefined)) {
    throw new RangeError('Invalid duration');
  }
  let total = 0n;
  for (let i = 0; i < MULTIPLIERS.length; i++) {
    const part = match[i + 1];
    if (part !== undefined) total += BigInt(part) * MULTIPLIERS[i];
  }
  if (total > BigInt(Number.MAX_SAFE_INTEGER)) {
    throw new RangeError('Duration exceeds Number.MAX_SAFE_INTEGER');
  }
  return Number(total);
}
