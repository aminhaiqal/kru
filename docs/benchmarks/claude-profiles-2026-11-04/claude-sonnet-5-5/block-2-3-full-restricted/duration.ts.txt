const UNITS = ['h', 'm', 's', 'ms'] as const;
const FACTORS: Record<(typeof UNITS)[number], number> = {
  h: 3_600_000,
  m: 60_000,
  s: 1_000,
  ms: 1,
};

export function parseDuration(text: string): number {
  if (typeof text !== 'string') throw new RangeError('Invalid duration');
  const re = /\s*(\d+)(ms|h|m|s)/y;
  const max = BigInt(Number.MAX_SAFE_INTEGER);
  let pos = 0;
  let lastIndex = -1;
  let total = 0n;
  let count = 0;
  while (true) {
    re.lastIndex = pos;
    const match = re.exec(text);
    if (!match) break;
    const unit = match[2] as (typeof UNITS)[number];
    const idx = UNITS.indexOf(unit);
    if (idx <= lastIndex) throw new RangeError('Invalid duration');
    lastIndex = idx;
    total += BigInt(match[1]) * BigInt(FACTORS[unit]);
    if (total > max) throw new RangeError('Duration too large');
    pos = re.lastIndex;
    count++;
  }
  if (count === 0 || !/^\s*$/.test(text.slice(pos))) {
    throw new RangeError('Invalid duration');
  }
  return Number(total);
}
